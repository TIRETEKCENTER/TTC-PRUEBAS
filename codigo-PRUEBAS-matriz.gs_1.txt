// ===== Reintento automático al guardar una fila (27/08/2026 — Baron revisó ErroresDoPost y
// encontró que, en horas pico, Google rechaza el guardado con "Demasiadas invocaciones
// simultáneas: Hojas de cálculo" — no es un bug de lógica, es que varios vendedores/técnicos
// guardan casi al mismo tiempo y Sheets tiene un límite de escrituras simultáneas. Antes esa
// excepción tronaba doPost_impl UNA sola vez, sin reintentar, y la fila se perdía para siempre
// aunque en la app pareciera que se había guardado bien. Se confirmó con las 10 terminaciones de
// Orden de Ejecución de llantas que Baron cerró el 26/08/2026 y que, pese a que la app no marcó
// error visible, nunca llegaron a la pestaña TerminacionesOS.
// Esta función reintenta el appendRow varias veces con una pausa creciente SOLO cuando el error
// es de este tipo transitorio (cupo de Google) — cualquier otro error (dato mal formado, pestaña
// faltante, etc.) se sigue reportando de inmediato a ErroresDoPost, sin reintentar, para no
// ocultar bugs reales. =====
function appendRowConReintento(sheet, fila) {
  var intentos = 0;
  var maxIntentos = 5;
  while (true) {
    try {
      sheet.appendRow(fila);
      return;
    } catch (err) {
      intentos++;
      var msg = String(err && err.message ? err.message : err);
      var esTransitorio = /invocaciones simultáneas|simultaneous invocations|too many requests|rate limit|internal error|timeout/i.test(msg);
      if (!esTransitorio || intentos >= maxIntentos) throw err;
      Utilities.sleep(400 * intentos + Math.floor(Math.random() * 300)); // espera creciente + variación aleatoria, para que dos escrituras que chocaron no se vuelvan a topar en el mismo instante
    }
  }
}

// ===== Convierte el valor de una celda de "Fecha" a texto dd/MM/yyyy sin importar cómo
// haya quedado guardada. Si la columna tiene formato "Automático" (el default de Sheets),
// al guardar un texto como "19/08/2026" con appendRow, Sheets lo detecta como fecha y lo
// convierte solo a un valor de tipo Date — al leerlo de vuelta con getValues() ya NO es el
// string "19/08/2026" sino un objeto Date, y compararlo contra "today" (String) nunca
// coincide aunque sea el mismo día. Por eso toda comparación de fecha pasa por aquí. =====
function normalizaFecha(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  }
  return String(v || '');
}

// ===== Mismo problema que normalizaFecha pero para la columna "Hora": si Sheets la guardó
// como valor de hora (typeof Date) en vez de texto, la regresa formateada HH:mm en vez de
// mandar el objeto Date crudo (que llegaba a la app como "1899-12-31T05:06:00.000Z"). =====
function normalizaHora(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'HH:mm');
  }
  return String(v || '');
}

// ===== Contador ATÓMICO de folios (22/08/2026 — corrige bug reportado por Baron: dos
// vendedores generando un vale casi al mismo tiempo recibían el MISMO número de folio, y sus
// pedidos/Órdenes de Ejecución terminaban mezclados en una sola tarjeta).
//
// El código viejo (ver abajo, ya no se usa) calculaba "next" CONTANDO cuántos folios de hoy ya
// existían en la hoja — pero solo LEÍA, nunca RESERVABA nada. El candado (LockService) se
// soltaba en cuanto terminaba esa lectura, antes de que el vendedor confirmara y se hiciera el
// POST real que escribe la fila. Si Baron y Oscar pedían folio casi al mismo tiempo, ambos
// veían "todavía no existe el folio 6 de hoy" y a los DOS se les daba el 6 — la vuelta de red
// entre "pedir folio" y "guardar el pedido" (fetchNextFolio -> finalizeVale en el frontend) es
// tiempo de sobra para que eso pase.
//
// La corrección: usar un contador guardado en PropertiesService (memoria propia de CADA
// proyecto de Apps Script — Matriz y Jacarandas ya tienen proyectos separados, así que sus
// contadores nunca se cruzan) que se lee Y se incrementa dentro del MISMO candado, en la misma
// operación — eso es lo que lo hace atómico: para el segundo vendedor que llega una fracción de
// segundo después, el contador YA quedó en 6, así que a él le toca el 7, sin importar si su
// pedido todavía no se ha guardado en la hoja.
function siguienteNumeroAtomico(clave) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var props = PropertiesService.getScriptProperties();
    var actual = parseInt(props.getProperty(clave) || '0', 10);
    if (isNaN(actual)) actual = 0;
    var siguiente = actual + 1;
    props.setProperty(clave, String(siguiente));
    return siguiente;
  } finally {
    lock.releaseLock();
  }
}

// ===== Devuelve el siguiente folio del día (formato PREFIJO+DDMMYY-N, ej. MAT310726-1),
// o la lista de pendientes de mostrador de hoy si viene ?action=pendientes =====
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'folio';
  if (action === 'pendientes') {
    return getPendientesHoy(e);
  }
  if (action === 'osHoy') {
    return getOSHoy(e);
  }
  if (action === 'folioDiagnostico') {
    return getFolioDiagnostico(e);
  }
  if (action === 'diagnosticosHoy') {
    return getDiagnosticosHoy(e);
  }
  if (action === 'cotizacionesHoy') {
    return getCotizacionesHoy(e);
  }
  if (action === 'ordenesServicioRefaccionesHoy') {
    return getOrdenesServicioRefaccionesHoy(e);
  }
  if (action === 'validacionesHoy') {
    return getValidacionesHoy(e);
  }
  if (action === 'tarifasMecanicos') {
    return getTarifasMecanicos(e);
  }
  if (action === 'ordenesServicioTodas') {
    return getOrdenesServicioTodas(e);
  }
  if (action === 'tarifasMontajeLlantas') {
    return getTarifasMontajeLlantas(e);
  }
  if (action === 'terminacionesOSTodas') {
    return getTerminacionesOSTodas(e);
  }
  // Conciliación de vales vs Alpha (27/10/2026) — ver bloque CONCILIACIÓN al final del archivo.
  if (action === 'valesDia') {
    return getValesDia(e);
  }
  if (action === 'conciliaciones') {
    return getConciliaciones(e);
  }
  // Pedidos de revendedores (31/10/2026) — ver bloque PEDIDOS DE REVENDEDORES al final.
  if (action === 'pedidosRevendedores') {
    return getPedidosRevendedores(e);
  }
  if (action === 'folioOSRDirecta') {
    // Antes este folio se armaba por completo en el navegador del vendedor, contando cuántas
    // Órdenes de Servicio Directa había en la lista de "hoy" que ya tenía descargada — sin
    // candado ni servidor de por medio. Mismo bug que el folio de pedidos, con margen de choque
    // todavía más grande. Ahora usa el mismo contador atómico (22/08/2026).
    var todayOSRD = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'ddMMyy');
    var nextOSRD = siguienteNumeroAtomico('folio_osrdirecta_' + todayOSRD);
    return ContentService.createTextOutput(JSON.stringify({ today: todayOSRD, next: nextOSRD }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'ddMMyy');
  var next = siguienteNumeroAtomico('folio_pedido_' + today);

  return ContentService.createTextOutput(JSON.stringify({ today: today, next: next }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Pendientes de mostrador de HOY: renglones de la hoja de pedidos con Status="Pendiente"
// y Fecha=hoy. Si viene ?vendedor=NOMBRE, se filtra solo a ese vendedor (vista del vendedor);
// sin ese parámetro trae TODOS (vista del administrador de almacén). =====
function getPendientesHoy(e) {
  var sheet = getPedidosSheet();
  var data = sheet.getDataRange().getValues();
  var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy');
  var vendedorFiltro = (e.parameter.vendedor || '').trim();

  var out = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var fecha = normalizaFecha(row[1]);
    var vendedor = String(row[3]);
    var status = String(row[11]);
    if (fecha !== today) continue;
    if (status !== 'Pendiente') continue;
    if (vendedorFiltro && vendedor !== vendedorFiltro) continue;
    out.push({
      // Mismo fix de fecha/hora sin normalizar que en getOSHoy (22/08/2026).
      folio: row[0], fecha: normalizaFecha(row[1]), hora: normalizaHora(row[2]), vendedor: row[3],
      clave: row[4], medida: row[5], marca: row[6], modelo: row[7],
      desc: row[8], cantidad: row[9], tipo: row[10], status: row[11],
      solicitudPara: row[12], sucursal: row[13], comentario: row[14]
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Órdenes de Servicio para reimprimir y para marcar terminado — y, desde el 27/08/2026,
// también la fuente del panel "Reporte de OS" con filtro de fecha/vendedor en el frontend.
// Antes se filtraba estricto a fecha=hoy (mismo bug que ya se había corregido en
// getDiagnosticosHoy/getCotizacionesHoy): en cuanto cruzaba la medianoche, una Orden de
// Ejecución de ayer que nadie había cerrado desaparecía de la lista y ya no había forma de
// marcarla terminado desde la app — que fue justo lo que le pasó a Baron/Sol el 25/08/2026 con
// 10 órdenes de llantas del día anterior. Luego (27/08/2026) se le quitó TODO filtro de fecha
// en el servidor: ahora regresa el histórico completo y es el frontend el que decide qué rango
// mostrar (Baron pidió poder filtrar el reporte por fecha y por vendedor). El frontend además
// ya solo consulta la sucursal del vendedor en sesión (nunca ambas), así nadie ve Órdenes de
// Servicio de la otra sucursal. =====
function getOSHoy(e) {
  var osSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('OrdenesServicio');
  if (!osSheet) {
    return ContentService.createTextOutput(JSON.stringify({ items: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var data = osSheet.getDataRange().getValues();

  var terminadosSet = {};
  var termSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('TerminacionesOS');
  if (termSheet) {
    var termData = termSheet.getDataRange().getValues();
    for (var t = 1; t < termData.length; t++) {
      var fTerm = String(termData[t][0] || '');
      if (fTerm) terminadosSet[fTerm] = true;
    }
  }

  var grouped = {};
  var order = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var folio = String(row[0] || '');
    if (!folio) continue;

    if (!grouped[folio]) {
      grouped[folio] = {
        folio: folio,
        pedidoFolioTxt: String(row[1] || ''),   // B - Folio(s) de pedido relacionado(s)
        // fecha/hora SIN normalizar (22/08/2026): si la columna quedó con formato de fecha/hora
        // "Automático" en Sheets, getValues() regresa un objeto Date de verdad, y
        // JSON.stringify() lo convierte solo con .toISOString() — de ahí el "1899-12-30T..."
        // que vio Baron en "Órdenes de Servicio — hoy" (1899-12-30 es el día cero de Sheets
        // para valores de solo-hora). normalizaFecha/normalizaHora ya existían para este mismo
        // problema en otras rutas; aquí se les había olvidado aplicar.
        fecha: normalizaFecha(row[2]), hora: normalizaHora(row[3]), vendedor: row[4], sucursal: row[5], referencia: row[6],
        telefono: row[7], marcaVehiculo: row[8], submarcaVehiculo: row[9], anioVehiculo: row[10],
        serviciosTxt: String(row[16] || ''),      // Q
        vehiculoTipoTxt: String(row[17] || ''),   // R
        posicionesTxt: String(row[18] || ''),     // S
        llantasViejasTxt: String(row[19] || ''),  // T
        status: row[20],                          // U
        montajeUbicacion: row[21],                // V
        candadoSeguridadTxt: String(row[22] || ''), // W
        comentariosEspeciales: row[23],           // X
        tecnico: row[24],                         // Y
        montajeRin: row[25] || '',                // Z
        tecnicoServicios: row[26] || '',          // AA
        clienteNombre: row[27] || '',              // AB
        items: []
      };
      order.push(folio);
    }
    grouped[folio].items.push({
      clave: row[11], medida: row[12], marca: row[13], modelo: row[14], cantidad: row[15]
    });
  }

  var out = order.map(function (f) { return grouped[f]; });
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Siguiente versión de cotización para un folio de diagnóstico dado. Cada vez que el
// cliente NO autoriza y el vendedor ajusta, se agrega una fila NUEVA con versión+1 — nunca se
// edita la fila anterior. Así queda el rastro completo de qué se propuso, qué se rechazó y qué
// se autorizó, sin trabajo extra para el vendedor (solo edita y reenvía). =====
function siguienteVersionCotizacion(folioDiagnostico) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('CotizacionesRefacciones');
  if (!sheet) return 1;
  var data = sheet.getDataRange().getValues();
  var maxV = 0;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(folioDiagnostico)) {
      var v = Number(data[i][1]) || 0;
      if (v > maxV) maxV = v;
    }
  }
  return maxV + 1;
}

// ===== Diagnósticos (TODOS, ya no filtrados por fecha ni por status — antes decía "de HOY" y
// en cuanto cruzaba la medianoche los pendientes de ayer desaparecían de la lista aunque
// siguieran sin cotizar. El filtro de "solo los que faltan por cotizar" ahora lo hace la propia
// pantalla en el navegador, NO el backend — así este mismo endpoint también sirve para jalar

  var vendedorFiltro = (e.parameter.vendedor || '').trim();
  var out = [];

    var row = data[i];
    if (!row[0]) continue;
    out.push({ mecanico: String(row[0]).trim(), porcentaje: Number(row[1]) || 0 });
  }
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Órdenes de Servicio de llantas (pestaña "OrdenesServicio") — SIN filtro de fecha, mismo
// motivo que las demás rutas "Todas/Hoy" del reporte de Sol: el rango de fechas lo decide la
// pantalla, no el backend. Se usa para calcular la comisión de montaje por rin. =====
function getOrdenesServicioTodas(e) {
  var osSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('OrdenesServicio');
  if (!osSheet) {
    return ContentService.createTextOutput(JSON.stringify({ items: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var data = osSheet.getDataRange().getValues();
  var grouped = {};
  var order = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var folio = String(row[0] || '');
    if (!folio) continue;
    if (!grouped[folio]) {
      grouped[folio] = {
        folio: folio,
        fecha: normalizaFecha(row[2]), hora: normalizaHora(row[3]), vendedor: row[4], sucursal: row[5],
        tecnico: row[24] || '', serviciosTxt: String(row[16] || ''), montajeRin: row[25] || '', items: []
      };
      order.push(folio);
    }
    grouped[folio].items.push({
      clave: row[11], medida: row[12], marca: row[13], modelo: row[14], cantidad: row[15]
    });
  }
  var out = order.map(function (f) { return grouped[f]; });
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== Envoltura de diagnóstico (26/08/2026, Baron reportó un montón de doPost fallando cada
// ~10 min sin poder ver el mensaje de error real desde la pantalla de Ejecuciones/Cloud Logs).
// Esto NO cambia el comportamiento normal — solo atrapa cualquier error dentro de doPost real
// (doPost_impl) y lo guarda en una pestaña "ErroresDoPost" del Sheet (se crea sola si no
// existe), con la fecha, el mensaje de error y el cuerpo crudo que llegó — así se puede ver el
// error exacto directo en el Sheet, sin depender de la consola de Apps Script. =====
function doPost(e) {
  try {
    return doPost_impl(e);
  } catch (err) {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var errSheet = ss.getSheetByName('ErroresDoPost');
      if (!errSheet) {
        errSheet = ss.insertSheet('ErroresDoPost');
        errSheet.appendRow(['Fecha/Hora', 'Error', 'Cuerpo recibido (postData)']);
      }
      var cuerpoRaw = (e && e.postData && e.postData.contents) ? e.postData.contents : '(sin postData -- e=' + JSON.stringify(e) + ')';
      errSheet.appendRow([
        Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss'),
        String(err && err.stack ? err.stack : err),
        cuerpoRaw.substring(0, 5000)
      ]);
    } catch (errLog) {
      // Si ni siquiera se pudo loggear el error, no hay nada más que hacer -- no debe tronar dos veces.
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

      data.tipoProducto || '',      // L - Tipo de producto (PCR/TBR/OTR/AGR/IND/MOTO/ATV/REMOLQUE/IMPLEMENTO/RIN/CAMARA/OTRO)
      data.costoNeto,               // M - Costo del producto (antes de IVA)
      data.fleteUnitario,           // N - Flete unitario usado en el cálculo (YA con IVA incluido)
      data.cantidad,                // O - Cantidad de piezas cotizadas
      data.precioUnitario,          // P - Precio unitario final (ya con margen e IVA aplicados)
      data.ivaUnitario,             // Q - IVA desglosado por pieza (informativo; $0 si es Agrícola)
      data.total,                   // R - Total (precio unitario x cantidad)
      data.telefonoCliente || '',   // S - Teléfono del cliente (obligatorio en el formulario)
      data.nombreCliente || '',     // T - Nombre del cliente/contacto (obligatorio en el formulario)
      data.empresaCliente || '',    // U - Nombre comercial de la empresa del cliente (opcional)
      data.medida || ''             // V - Medida (ancho/serie/rin) — agregada 11/09/2026
    ]);
  } else if (data.docType === 'terminacionOS') {
    // Terminación de la Orden de Ejecución de llantas — el vendedor confirma con el técnico
    // qué se hizo y qué no, con razón si algo quedó pendiente. En cuanto existe esta fila
    // para un Folio, ese trabajo queda cerrado.
    // 27/08/2026: agregado LockService + appendRowConReintento — este era el ÚNICO guardado
    // "sensible" (bloquea el pago de comisión del mecánico y libera el folio) que no tenía
    // candado ni reintento. Baron cerró 10 Órdenes el 26/08/2026 en hora pico y todas se
    // perdieron con el error de Google "Demasiadas invocaciones simultáneas" sin ningún aviso
    // en la app — se confirmó revisando la pestaña ErroresDoPost.
    var lock8 = LockService.getScriptLock();
    lock8.waitLock(15000);
    try {
      var termSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('TerminacionesOS');
      if (!termSheet) {
        return ContentService.createTextOutput(JSON.stringify({status: 'error', message: 'Falta la pestaña TerminacionesOS'}))
          .setMimeType(ContentService.MimeType.JSON);
      }
      appendRowConReintento(termSheet, [
        data.folio,                 // A - Folio de la Orden de Ejecución (OrdenesServicio)
        data.fecha,                  // B - Fecha
        data.hora,                   // C - Hora
        data.vendedorConfirma || '', // D - Vendedor que confirma la terminación
        data.sucursal || '',         // E - Sucursal
        data.tecnico || '',          // F - Técnico que realizó el montaje
        data.tecnicoServicios || '', // G - Técnico de servicios (si fue distinto)
        JSON.stringify(data.itemsValidados || []), // H - Llantas/servicios confirmados, con completado true/false
        data.comentarios || ''       // I - Comentarios / razón de lo no concluido
      ]);
    } finally {
      lock8.releaseLock();
    }
  } else if (data.docType === 'statusUpdate') {
    // Busca filas existentes del pedido original (por Folio + Clave) y les actualiza la
    // columna Status (L). Se usa cuando el vendedor decide, al ligar pedidos, qué llanta se
    // instaló y cuál se devolvió a almacén.
    var lock2 = LockService.getScriptLock();
    lock2.waitLock(10000);
    try {
      var sheet2 = getPedidosSheet();
      var values = sheet2.getDataRange().getValues();
      (data.updates || []).forEach(function(u) {
        for (var i = 1; i < values.length; i++) {
          if (String(values[i][0]) === String(u.folio) && String(values[i][4]) === String(u.clave)) {
            sheet2.getRange(i + 1, 12).setValue(u.status); // columna L = Status
          }
        }
      });
    } finally {
      lock2.releaseLock();
    }
  } else {
    var sheet = getPedidosSheet();
    data.items.forEach(function(item) {
      appendRowConReintento(sheet, [
        data.folio,         // A - Folio
        data.fecha,         // B - Fecha
        data.hora,          // C - Hora
        data.vendedor,      // D - Vendedor
        item.clave,         // E - Clave
        item.medida,        // F - Medida
        item.marca,         // G - Marca
        item.modelo,        // H - Modelo
        item.desc,          // I - Descripción
        item.cantidad,      // J - Cantidad
        item.tipo,          // K - Tipo
        'Pendiente',        // L - Status
        data.solicitudPara, // M - Solicitud Para
        data.sucursal,       // N - Sucursal (Matriz / Jacarandas)
        data.comentario || '', // O - Referencia (cliente / vehículo / orden)
        data.clienteNombre || '', // P - Nombre del cliente (obligatorio en ventas)
        data.clienteWhatsapp || '' // Q - WhatsApp del cliente (obligatorio en ventas)
      ]);
    });
  }

  return ContentService.createTextOutput(JSON.stringify({status: 'ok'}))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== RESPALDO DIARIO: copia completa de este Sheet a Drive =====
// Guarda una copia fechada de TODO el archivo (todas las pestañas) en una carpeta de Drive
// llamada "Respaldos Urbiola Llantas". Protege contra ediciones o borrados accidentales.
function respaldoDiario() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var folderName = 'Respaldos Urbiola Llantas';
  var folders = DriveApp.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
  var fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  var nombreRespaldo = ss.getName() + ' - Respaldo ' + fecha;
  var archivoOriginal = DriveApp.getFileById(ss.getId());
  archivoOriginal.makeCopy(nombreRespaldo, folder);
}

// ===== Instala el disparador diario. EJECUTAR MANUALMENTE UNA SOLA VEZ =====
// (▶ Ejecutar, en el editor de Apps Script, con la función "crearTriggerRespaldoDiario"
// seleccionada). Hay que correrlo por separado en CADA hoja (Matriz y Jacarandas), porque
// cada una tiene su propio proyecto de Apps Script. El respaldo corre todos los días a las
// 23:00 (hora del script). Si ya existía un disparador de respaldoDiario, lo borra primero.
function crearTriggerRespaldoDiario() {
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === 'respaldoDiario') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('respaldoDiario')
    .timeBased()
    .everyDays(1)
    .atHour(23)
    .create();
}

// ===== CONCILIACIÓN DE VALES vs FACTURAS/REMISIONES DE ALPHA (27/10/2026, Ricardo) =====
// Propuesta completa: documento "Propuesta: Conciliación de Vales vs Facturas y Remisiones".
// El vendedor sube a la app sus reportes de Alpha del día anterior (sacados con SU usuario);
// la app los cruza contra los vales de este Sheet. Aquí solo hay dos cosas:
//   1) GET ?action=valesDia&fecha=dd/MM/yyyy[&vendedor=NOMBRE]  -> vales de ese día (pestaña de
//      pedidos, la primera). Sin filtrar por tipo de solicitud ni status: esas reglas (Muestra
//      en piso nunca se concilia, cancelados fuera) las aplica la pantalla.
//   2) GET ?action=conciliaciones[&fecha=dd/MM/yyyy]  +  POST docType 'conciliacion' -> pestaña
//      "Conciliaciones" (se crea sola). Solo se AGREGAN filas (nunca se edita una fila guardada):
//      cada explicación de diferencia y cada cierre de día es una fila; la más reciente manda.
// ============================================================================
var CONC_SHEET_NAME = 'Conciliaciones';
var CONC_HEADERS = ['FechaRegistro', 'HoraRegistro', 'DiaConciliado', 'Vendedor', 'Sucursal', 'Tipo',
                    'FolioVale', 'Documento', 'Clave', 'Estatus', 'Motivo', 'Comentario', 'DatosJSON'];
function getConcSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(CONC_SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(CONC_SHEET_NAME);
    sh.appendRow(CONC_HEADERS);
    sh.getRange('A:C').setNumberFormat('@'); // fecha/hora/día como TEXTO: evita el bug "1899-12-30"
  }
  return sh;
}
function getValesDia(e) {
  var fecha = String(e.parameter.fecha || '').trim();
  var vendedorFiltro = String(e.parameter.vendedor || '').trim();
  var data = getPedidosSheet().getDataRange().getValues();
  var out = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0]) continue;
    if (fecha && normalizaFecha(row[1]) !== fecha) continue;
    if (vendedorFiltro && String(row[3]).trim() !== vendedorFiltro) continue;
    out.push({
      fila: i + 1,
      folio: String(row[0]), fecha: normalizaFecha(row[1]), hora: normalizaHora(row[2]),
      vendedor: String(row[3] || ''), clave: String(row[4] == null ? '' : row[4]),
      medida: row[5], marca: row[6], modelo: row[7], desc: row[8],
      cantidad: Number(row[9]) || 0, tipo: row[10], status: String(row[11] || ''),
      solicitudPara: String(row[12] || ''), sucursal: String(row[13] || ''),
      comentario: String(row[14] || ''),
      // P/Q/R: según la época del renglón aquí quedó el nombre del cliente / WhatsApp. Se mandan
      // las tres tal cual y la pantalla usa la primera que traiga texto.
      colP: String(row[15] || ''), colQ: String(row[16] || ''), colR: String(row[17] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}
function getConciliaciones(e) {
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONC_SHEET_NAME);
  if (!sh) {
    return ContentService.createTextOutput(JSON.stringify({ items: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var fecha = String((e && e.parameter && e.parameter.fecha) || '').trim();
  var data = sh.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    if (!r[0]) continue;
    var dia = normalizaFecha(r[2]);
    if (fecha && dia !== fecha) continue;
    var datos = {};
    try { datos = JSON.parse(r[12] || '{}'); } catch (err) { datos = {}; }
    out.push({
      fechaRegistro: normalizaFecha(r[0]), horaRegistro: normalizaHora(r[1]), dia: dia,
      vendedor: String(r[3] || ''), sucursal: String(r[4] || ''), tipo: String(r[5] || ''),
      folioVale: String(r[6] || ''), documento: String(r[7] || ''), clave: String(r[8] || ''),
      estatus: String(r[9] || ''), motivo: String(r[10] || ''), comentario: String(r[11] || ''),
      datos: datos
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ items: out }))
    .setMimeType(ContentService.MimeType.JSON);
}
function guardarConciliacion(data) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var sh = getConcSheet();
    var ahora = new Date();
    var tz = Session.getScriptTimeZone();
    (data.filas || [data]).forEach(function (f) {
      appendRowConReintento(sh, [
        Utilities.formatDate(ahora, tz, 'dd/MM/yyyy'),   // A - Fecha de registro
        Utilities.formatDate(ahora, tz, 'HH:mm'),        // B - Hora de registro
        f.dia || data.dia || '',                         // C - Día conciliado (dd/MM/yyyy)
        f.vendedor || data.vendedor || '',               // D - Vendedor
        f.sucursal || data.sucursal || '',               // E - Sucursal
        f.tipo || '',                                    // F - 'explicacion' | 'cierre' | 'revisado'
        f.folioVale || '',                               // G - Folio del vale (si aplica)
        f.documento || '',                               // H - Documento de Alpha (ej. F 71539, Rem 6141)
        f.clave || '',                                   // I - Clave de producto
        f.estatus || '',                                 // J - Estatus del cruce (sin_documento, medida_cambiada...)
        f.motivo || '',                                  // K - Motivo elegido de la lista
        f.comentario || '',                              // L - Comentario libre
        JSON.stringify(f.datos || {})                    // M - Detalle (resumen del día, renglones del cruce)
      ]);
    });
  } finally {
    lock.releaseLock();
  }
}

  if (!sh) {
    sh = ss.insertSheet(REV_SHEET_NAME);

    sh.getRange('B:C').setNumberFormat('@'); // fecha/hora como texto (bug "1899-12-30")

function guardarPedidoRevendedor(data) {
  var tz = Session.getScriptTimeZone();
  var ahora = new Date();
  var hoy = Utilities.formatDate(ahora, tz, 'ddMMyy');
  var folio = 'REV' + hoy + '-' + siguienteNumeroAtomico('folio_revendedor_' + hoy);
  var fecha = Utilities.formatDate(ahora, tz, 'dd/MM/yyyy');
  var hora = Utilities.formatDate(ahora, tz, 'HH:mm');
  var sh = getRevSheet();
  (data.items || []).forEach(function (it) {
    appendRowConReintento(sh, [
      folio, fecha, hora,
      data.revendedor || '', data.negocio || '', data.vendedorAsignado || '',
      it.clave || '', it.medida || '', it.marca || '', it.modelo || '', it.desc || '',
      Number(it.cantidad) || 0,
      it.precio === '' || it.precio == null ? 'Consultar' : Number(it.precio),
      it.precio === '' || it.precio == null ? '' : Number(it.precio) * (Number(it.cantidad) || 0),
      it.disponibilidad || '', 'Pendiente', data.comentario || ''
    ]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', folio: folio, fecha: fecha, hora: hora }))
    .setMimeType(ContentService.MimeType.JSON);
}
function getPedidosRevendedores(e) {
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(REV_SHEET_NAME);
  if (!sh) return ContentService.createTextOutput(JSON.stringify({ items: [] })).setMimeType(ContentService.MimeType.JSON);
  var vend = String((e && e.parameter && e.parameter.vendedor) || '').trim();
  var data = sh.getDataRange().getValues(), out = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    if (!r[0]) continue;
    if (vend && String(r[5]) !== vend) continue;
    out.push({ folio: r[0], fecha: normalizaFecha(r[1]), hora: normalizaHora(r[2]), revendedor: r[3], negocio: r[4],
               vendedorAsignado: r[5], clave: String(r[6]), medida: r[7], marca: r[8], modelo: r[9], desc: r[10],
               cantidad: r[11], precio: r[12], importe: r[13], disponibilidad: r[14], status: r[15], comentario: r[16] });
  }
  return ContentService.createTextOutput(JSON.stringify({ items: out })).setMimeType(ContentService.MimeType.JSON);
}
// Cambio de status de un pedido de revendedor (todas sus filas, por folio) desde la app de
// vendedores: "Pendiente" -> "Atendido" / "Cancelado". Se agrega quién y cuándo en Comentario.
function actualizarStatusPedidoRevendedor(data) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sh = getRevSheet();
    var vals = sh.getDataRange().getValues();
    var nota = (data.status || '') + ' por ' + (data.quien || '') + ' ' +
               Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
    for (var i = 1; i < vals.length; i++) {
      if (String(vals[i][0]) !== String(data.folio)) continue;
      sh.getRange(i + 1, 16).setValue(data.status || 'Atendido');                         // P - Status
      var com = String(vals[i][16] || '');
      sh.getRange(i + 1, 17).setValue(com ? com + ' | ' + nota : nota);                   // Q - Comentario
    }
  } finally {
    lock.releaseLock();
  }
}
